-- ============================================================
-- EXAMCENTER UNIVERSAL DATABASE SCHEMA — AUTHORITATIVE REFERENCE
-- DATABASE: cbt_app_db
-- ENGINE: MySQL / MariaDB
-- CHARSET: utf8mb4
-- ============================================================
--
-- This file documents the DATABASE SCHEMA ACTUALLY USED by the current
-- Examcenter runtime.
-- ARCHITECTURE
--   1. Legacy Core CBT tables remain for backward compatibility.
--   2. Universal tables provide configurable institution/unit/programme/
--      level/period/course/person/assessment relationships.
--   3. Theory Assessment is an additive assessment capability.
--   4. Pro Assessment Suite is additive and entitlement-gated.
--   5. Core exam delivery remains usable offline and is not Pro-locked.
--
-- CANONICAL UNIVERSAL COLUMN NAMES
--   institutions: name, code, is_active
--   institution_types: code, name, is_active
--   organizational_units: parent_id, code, name, is_active
--   organizational_unit_types: code, name, is_active
--   academic_periods: parent_id, code, name, starts_at, ends_at
--   programmes: code, name, is_active
--   programme_levels: organizational_unit_id, code, name, sort_order
--   people: institution_id, display_name, external_ref, is_active
--   institution_memberships: institution_id, person_id, role_code, status
--   unit_memberships: unit_id, person_id, membership_role, status
--   courses: programme_id, code, name, is_active
--   course_assignments: course_id, unit_id, person_id, academic_period_id,
--                       role_code
--   assessment_groups: code, name, group_type, weighting
--   assessment_group_assignments: unit_id, programme_id,
--                                  academic_period_id, course_id
--   tests: institution_id, organizational_unit_id, programme_id,
--          programme_level_id, academic_period_id, course_id,
--          assessment_group_id
--
-- LEGACY COMPATIBILITY
--   Retired draft Universal names such as institution_name, institution_code,
--   unit_name, unit_code, period_name, period_code, parent_unit_id and
--   institution_membership_id are NOT canonical runtime names. Migration
--   20261007_0005 reads old installations and backfills canonical fields.
--
-- ============================================================

    completeness_score DECIMAL(5,4) NULL,

    explanation LONGTEXT NULL,
    strengths LONGTEXT NULL,
    weaknesses LONGTEXT NULL,
    missing_points LONGTEXT NULL,
    grading_model VARCHAR(150) NULL,
    grading_model_version VARCHAR(100) NULL,
    grading_prompt_version VARCHAR(50) NULL,
    status VARCHAR(30) NOT NULL DEFAULT 'ai_provisional',
    graded_at DATETIME NULL,
    reviewed_by INT NULL,
    reviewed_at DATETIME NULL,
    review_reason TEXT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    UNIQUE KEY uq_theory_mark_answer (answer_id),
    CONSTRAINT fk_theory_mark_answer FOREIGN KEY (answer_id) REFERENCES theory_answers(id) ON DELETE CASCADE,

    CHECK (awarded_marks >= 0),
    CHECK (awarded_marks <= maximum_marks),
    CHECK (percentage >= 0 AND percentage <= 100)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
